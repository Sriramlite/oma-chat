-- EXPERIMENT 1: DDL COMMANDS
CREATE DATABASE IF NOT EXISTS ex1_ddl;
USE ex1_ddl;
DROP TABLE IF EXISTS stud;
CREATE TABLE stud (sname VARCHAR(30), sid VARCHAR(10), sage INT, sarea VARCHAR(20));
SHOW TABLES;
DESCRIBE stud;
ALTER TABLE stud ADD COLUMN phoneno BIGINT AFTER sage;
DESCRIBE stud;
ALTER TABLE stud MODIFY COLUMN sage VARCHAR(20);
DESCRIBE stud;
ALTER TABLE stud DROP COLUMN sage;
DESCRIBE stud;
TRUNCATE TABLE stud;
DESCRIBE stud;
DROP TABLE stud;
SHOW TABLES;
