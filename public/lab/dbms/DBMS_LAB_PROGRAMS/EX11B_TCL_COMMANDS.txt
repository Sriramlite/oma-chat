-- EXPERIMENT 11B: TCL COMMANDS
CREATE DATABASE IF NOT EXISTS ex11b_tcl;
USE ex11b_tcl;
DROP TABLE IF EXISTS emp;
CREATE TABLE emp (
 ename VARCHAR(20),eid INT PRIMARY KEY,salary INT,exp INT
) ENGINE=InnoDB;
INSERT INTO emp VALUES ('kavi',1,40000,7),('kar',2,30000,6),('shanthi',3,40000,9);
SELECT * FROM emp;
START TRANSACTION;
SAVEPOINT ee;
INSERT INTO emp VALUES ('uma',6,70000,15);
SELECT * FROM emp;
ROLLBACK TO SAVEPOINT ee;
SELECT * FROM emp;
COMMIT;
SELECT * FROM emp;
