-- EXPERIMENT 6: SUBQUERIES
CREATE DATABASE IF NOT EXISTS ex6_subqueries;
USE ex6_subqueries;
DROP TABLE IF EXISTS sworks;
DROP TABLE IF EXISTS spros;
DROP TABLE IF EXISTS sempls;
DROP TABLE IF EXISTS sdments;
CREATE TABLE sempls (
 eno INT PRIMARY KEY, ename VARCHAR(20), edob VARCHAR(15), gender VARCHAR(10),
 doj VARCHAR(20), desg VARCHAR(30), bpay DECIMAL(10,2), dno INT
);

-- Query 1: employees earning below average pay
SELECT * FROM sempls WHERE bpay<(SELECT AVG(bpay) FROM sempls);
-- Query 2: employee count by department
SELECT dno,COUNT(eno) AS employee_count FROM sempls GROUP BY dno;
-- Query 3: salary range
SELECT * FROM sempls WHERE bpay BETWEEN 10000 AND 20000;
-- Query 4: projects controlled by CSE department
SELECT e.eno,e.ename,d.dname,p.pno,p.pname
FROM sempls e JOIN sdments d ON d.dno=e.dno
JOIN spros p ON p.dno=d.dno WHERE d.dname='cse';
-- Query 5: employees with more than four projects on a day
SELECT e.eno,e.ename,e.dno,w.datework
FROM sempls e JOIN sworks w ON w.eno=e.eno
JOIN (SELECT eno,datework FROM sworks GROUP BY eno,datework
      HAVING COUNT(DISTINCT pno)>4) x
ON x.eno=w.eno AND x.datework=w.datework
GROUP BY e.eno,e.ename,e.dno,w.datework;
