-- EXPERIMENT 4: SQL OPERATORS
CREATE DATABASE IF NOT EXISTS ex4_operators;
USE ex4_operators;
DROP TABLE IF EXISTS numbers;
CREATE TABLE numbers (num1 INT, num2 INT);
INSERT INTO numbers VALUES (5,10),(15,20);
SELECT num1,num2,num1+num2 AS addition FROM numbers;
SELECT num1,num2,num1-num2 AS subtraction FROM numbers;
SELECT num1,num2,num1*num2 AS multiplication FROM numbers;
SELECT num1,num2,num1/num2 AS division FROM numbers;
SELECT num1,num2,MOD(num1,num2) AS modulus FROM numbers;
SELECT num1,num2,POW(num1,2) AS square_of_num1 FROM numbers;
DROP TABLE IF EXISTS TECSE;
CREATE TABLE TECSE (ROLL INT PRIMARY KEY, NAME VARCHAR(30), DBMS INT, JAVA INT);
INSERT INTO TECSE VALUES
(1,'KUMBHAR',15,25),(2,'MALI',40,45),(3,'SHELKE',35,22),
(4,'THORAT',50,42),(5,'MORE',45,48);
SELECT * FROM TECSE;
SELECT ROLL,NAME FROM TECSE WHERE DBMS>30 OR JAVA>35;
SELECT ROLL,NAME FROM TECSE WHERE DBMS>30 AND JAVA>32;
SELECT ROLL,NAME FROM TECSE WHERE DBMS>30;
SELECT ROLL,NAME FROM TECSE WHERE DBMS<>JAVA;
SELECT ROLL,NAME FROM TECSE WHERE DBMS>=40;
SELECT ROLL,NAME FROM TECSE WHERE JAVA<=25;
