-- Q19: LIBRARY MANAGEMENT TABLES AND RELATIONSHIPS
CREATE DATABASE IF NOT EXISTS q19_library;
USE q19_library;
DROP TABLE IF EXISTS issues;
DROP TABLE IF EXISTS books;
DROP TABLE IF EXISTS members;
CREATE TABLE books (
 book_id INT PRIMARY KEY, title VARCHAR(100) NOT NULL,
 author VARCHAR(80), isbn VARCHAR(20) UNIQUE
);
CREATE TABLE members (
 member_id INT PRIMARY KEY, member_name VARCHAR(60) NOT NULL,
 phone VARCHAR(20) UNIQUE
);
CREATE TABLE issues (
 issue_id INT PRIMARY KEY, book_id INT NOT NULL,
 member_id INT NOT NULL, issue_date DATE NOT NULL,
 return_date DATE NULL,
 FOREIGN KEY(book_id) REFERENCES books(book_id),
 FOREIGN KEY(member_id) REFERENCES members(member_id)
);
