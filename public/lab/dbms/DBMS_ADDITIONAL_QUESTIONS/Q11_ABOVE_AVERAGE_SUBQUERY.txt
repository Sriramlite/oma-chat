-- Q11: EMPLOYEES WHOSE SALARY IS ABOVE THE AVERAGE
CREATE DATABASE IF NOT EXISTS q11_subquery;
USE q11_subquery;
DROP TABLE IF EXISTS employees;
CREATE TABLE employees (
 emp_id INT PRIMARY KEY, emp_name VARCHAR(50),
 department VARCHAR(40), salary DECIMAL(10,2)
);
INSERT INTO employees VALUES
(1,'Arun','IT',45000),(2,'Bala','HR',32000),
(3,'Charan','IT',60000),(4,'Divya','Sales',38000),
(5,'Esha','HR',52000);
SELECT emp_id,emp_name,department,salary
FROM employees
WHERE salary>(SELECT AVG(salary) FROM employees);
